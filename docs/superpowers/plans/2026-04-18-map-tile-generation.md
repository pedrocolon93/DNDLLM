# Map Tile Generation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the flat floor + random POI map with a room+corridor layout that generates style-consistent, adjacency-aware tile textures using Gemini's multimodal image API as a style anchor.

**Architecture:** A style anchor (floor tile) is generated first via `GenerateImage`. Every subsequent tile type is generated via a new `GenerateStyledTile(prompt, anchor, cacheKey)` method that sends the anchor as a base64 image alongside the text prompt in a Gemini multimodal request. Transition tiles are generated by `GenerateTransitionTile(prompt, texA, texB, cacheKey)` which sends two reference images. `MapGenerator` owns all prompt templates and tile type logic; `LLMService` only handles HTTP calls — no cross-namespace imports added.

**Tech Stack:** Unity C#, OpenRouter API (Gemini multimodal via `google/gemini-*`), `UnityWebRequest`, `Task.WhenAll` for parallel generation, `Dictionary<TileType, Texture2D>` for tile texture lookup.

```csharp
        private void SetTile(int x, int y, TileType type)
        {
            var d = TileDefaults.For(type);
            grid[x, y].type = type;
            grid[x, y].walkable = d.walkable;
            grid[x, y].blocksVision = d.blocksVision;
        }

        private bool IsAdjacentTo(int x, int y, TileType type)
        {
            int[] dx = { 1, -1, 0, 0 };
            int[] dy = { 0, 0, 1, -1 };
            for (int d = 0; d < 4; d++)
            {
                int nx = x + dx[d], ny = y + dy[d];
                if (nx >= 0 && nx < width && ny >= 0 && ny < height && grid[nx, ny].type == type)
                    return true;
            }
            return false;
        }

        public MapTile GetTile(int x, int y)
        {
            if (x < 0 || x >= width || y < 0 || y >= height) return null;
            return grid[x, y];
        }
```

```csharp
        private Texture2D GetTileTexture(int x, int y)
        {
            TileType myType = grid[x, y].type;
            int[] ddx = { 1, -1, 0, 0 };
            int[] ddy = { 0, 0, 1, -1 };
            for (int d = 0; d < 4; d++)
            {
                int nx = x + ddx[d], ny = y + ddy[d];
                if (nx < 0 || nx >= width || ny < 0 || ny >= height) continue;
                TileType neighborType = grid[nx, ny].type;
                if (neighborType == myType) continue;
                var pair = myType < neighborType ? (myType, neighborType) : (neighborType, myType);
                if (transitionTextures.ContainsKey(pair))
                    return transitionTextures[pair];
            }
            return tileTextures.ContainsKey(myType) ? tileTextures[myType] : Texture2D.whiteTexture;
        }
```

```csharp
        private void GenerateLayout()
        {
            // 1. Fill entire grid with Wall
            for (int x = 0; x < width; x++)
                for (int y = 0; y < height; y++)
                    SetTile(x, y, TileType.Wall);

            // 2. Place 2–3 rectangular rooms
            var rooms = new List<RectInt>();
            int roomCount = Random.Range(2, 4);
            for (int i = 0; i < roomCount; i++)
            {
                int rw = Random.Range(3, 5);
                int rh = Random.Range(3, 5);
                int rx = Random.Range(1, width - rw - 1);
                int ry = Random.Range(1, height - rh - 1);
                var room = new RectInt(rx, ry, rw, rh);
                rooms.Add(room);
                for (int cx = room.xMin; cx < room.xMax; cx++)
                    for (int cy = room.yMin; cy < room.yMax; cy++)
                        SetTile(cx, cy, TileType.Floor);
            }

            // 3. Connect room pairs with L-shaped corridors
            for (int i = 0; i < rooms.Count - 1; i++)
                CarveCorridor(rooms[i].center, rooms[i + 1].center);

            // 4. Add a 2x2 water pool (only overwrites Wall tiles)
            int wx = Random.Range(1, width - 3);
            int wy = Random.Range(1, height - 3);
            for (int x = wx; x < wx + 2; x++)
                for (int y = wy; y < wy + 2; y++)
                    if (grid[x, y].type == TileType.Wall)
                        SetTile(x, y, TileType.Water);

            // 5. Place Doors at Path tiles on room boundaries
            PlaceDoors();

            // 6. Place POIs inside rooms
            PlacePOIs(rooms);
        }

        private void CarveCorridor(Vector2Int a, Vector2Int b)
        {
            int x = a.x;
            while (x != b.x)
            {
                if (grid[x, a.y].type == TileType.Wall)
                    SetTile(x, a.y, TileType.Path);
                x += (b.x > a.x) ? 1 : -1;
            }
            int y = a.y;
            while (y != b.y)
            {
                if (grid[b.x, y].type == TileType.Wall)
                    SetTile(b.x, y, TileType.Path);
                y += (b.y > a.y) ? 1 : -1;
            }
        }

        private void PlaceDoors()
        {
            for (int x = 1; x < width - 1; x++)
                for (int y = 1; y < height - 1; y++)
                    if (grid[x, y].type == TileType.Path
                        && IsAdjacentTo(x, y, TileType.Floor)
                        && IsAdjacentTo(x, y, TileType.Wall))
                        SetTile(x, y, TileType.Door);
        }

        private void PlacePOIs(List<RectInt> rooms)
        {
            TileType[] poiTypes = { TileType.Chest, TileType.EnemySpawn, TileType.Exit };
            for (int i = 0; i < Mathf.Min(rooms.Count, poiTypes.Length); i++)
            {
                var room = rooms[i];
                int px = Random.Range(room.xMin, room.xMax);
                int py = Random.Range(room.yMin, room.yMax);
                SetTile(px, py, poiTypes[i]);
            }
        }
```

```csharp
        private string BuildMultimodalRequest(string model, List<string> base64Images, string textPrompt)
        {
            // JsonUtility cannot serialize mixed-type content arrays, so JSON is built manually.
            var imageEntries = new System.Text.StringBuilder();
            foreach (string b64 in base64Images)
            {
                if (imageEntries.Length > 0) imageEntries.Append(",");
                imageEntries.Append(
                    $"{{\"type\":\"image_url\",\"image_url\":{{\"url\":\"data:image/png;base64,{b64}\"}}}}"
                );
            }
            string escaped = textPrompt
                .Replace("\\", "\\\\")
                .Replace("\"", "\\\"")
                .Replace("\n", "\\n");
            return
                $"{{\"model\":\"{model}\"," +
                $"\"messages\":[{{\"role\":\"user\",\"content\":[{imageEntries}," +
                $"{{\"type\":\"text\",\"text\":\"{escaped}\"}}]}}]," +
                $"\"modalities\":[\"image\",\"text\"]}}";
        }
```

```csharp
        private async Task<Texture2D> SendGeminiImageRequest(string jsonBody)
        {
            string chatUrl = "https://openrouter.ai/api/v1/chat/completions";
            using (var request = new UnityEngine.Networking.UnityWebRequest(chatUrl, "POST"))
            {
                byte[] bodyRaw = System.Text.Encoding.UTF8.GetBytes(jsonBody);
                request.uploadHandler = new UnityEngine.Networking.UploadHandlerRaw(bodyRaw);
                request.downloadHandler = new UnityEngine.Networking.DownloadHandlerBuffer();
                request.SetRequestHeader("Content-Type", "application/json");
                request.SetRequestHeader("Authorization", $"Bearer {apiKey}");
                request.SetRequestHeader("HTTP-Referer", "https://github.com/google-deepmind/antigravity");
                request.SetRequestHeader("X-Title", "DNDLLM");

                var operation = request.SendWebRequest();
                while (!operation.isDone) await Task.Yield();

                if (request.result == UnityEngine.Networking.UnityWebRequest.Result.Success)
                {
                    Debug.Log($"Gemini Multimodal Response: {request.downloadHandler.text}");
                    var responseData = JsonUtility.FromJson<OpenRouterResponse>(request.downloadHandler.text);
                    if (responseData?.choices?.Count > 0)
                    {
                        var msg = responseData.choices[0].message;
                        if (msg.images != null && msg.images.Count > 0)
                            return ParseBase64Image(msg.images[0].image_url.url);
                        return ParseBase64Image(msg.content);
                    }
                }
                else
                {
                    Debug.LogError($"Gemini Image Error: {request.error} - {request.downloadHandler.text}");
                }
            }
            return null;
        }
```

```csharp
        public async Task<Texture2D> GenerateStyledTile(string prompt, Texture2D styleAnchor, string cacheKey)
        {
            if (useCache)
            {
                Texture2D cached = ImageCache.Load(cacheKey);
                if (cached != null)
                {
                    Debug.Log($"[LLMService] Cache hit: {cacheKey}");
                    return cached;
                }
            }
            if (useMock)
            {
                await Task.Delay(500);
                return CreateMockTexture();
            }
            if (!imageModel.ToLower().StartsWith("google/"))
            {
                // Non-Gemini fallback: text-only generation, no style anchor
                return await GenerateImage(prompt);
            }

            string anchorB64 = System.Convert.ToBase64String(styleAnchor.EncodeToPNG());
            string fullPrompt = prompt + " — exact same art style as the reference image.";
            string json = BuildMultimodalRequest(imageModel, new List<string> { anchorB64 }, fullPrompt);

            Texture2D result = await SendGeminiImageRequest(json);
            if (result != null && useCache) ImageCache.Save(cacheKey, result);
            return result;
        }
```

```csharp
        public async void GenerateMap(string keywords)
        {
            Debug.Log($"Generating map with keywords: {keywords}...");

            foreach (Transform child in transform)
                Destroy(child.gameObject);
            tileTextures.Clear();
            transitionTextures.Clear();

            string primaryKeyword = string.IsNullOrEmpty(keywords) ? "dungeon" : keywords.Split(',')[0].Trim();

            // 1. Build the layout (fills grid with tile types and walkability)
            grid = new MapTile[width, height];
            for (int x = 0; x < width; x++)
                for (int y = 0; y < height; y++)
                    grid[x, y] = new MapTile { x = x, y = y };
            GenerateLayout();

            // 2. Generate style anchor — the Floor tile, generated as plain text prompt
            string floorPrompt = TilePromptFor(TileType.Floor, primaryKeyword);
            Texture2D styleAnchor = await LLMService.Instance.GenerateImage(floorPrompt);
            if (styleAnchor == null) styleAnchor = Texture2D.whiteTexture;
            tileTextures[TileType.Floor] = styleAnchor;

            // 3. Collect unique tile types present in the layout (excluding Floor)
            var uniqueTypes = new HashSet<TileType>();
            for (int x = 0; x < width; x++)
                for (int y = 0; y < height; y++)
                    uniqueTypes.Add(grid[x, y].type);
            uniqueTypes.Remove(TileType.Floor);

            // 4. Generate one styled texture per unique type — parallel, using style anchor
            var typeTasks = new Dictionary<TileType, Task<Texture2D>>();
            foreach (TileType type in uniqueTypes)
            {
                string cacheKey = $"{type}:{primaryKeyword}";
                string prompt = TilePromptFor(type, primaryKeyword);
                typeTasks[type] = LLMService.Instance.GenerateStyledTile(prompt, styleAnchor, cacheKey);
            }
            await Task.WhenAll(typeTasks.Values);
            foreach (var kvp in typeTasks)
                tileTextures[kvp.Key] = kvp.Value.Result ?? Texture2D.whiteTexture;

            // 5. Collect unique order-independent adjacency pairs
            var adjPairs = new HashSet<(TileType, TileType)>();
            int[] ddx = { 1, 0 };
            int[] ddy = { 0, 1 };
            for (int x = 0; x < width; x++)
            {
                for (int y = 0; y < height; y++)
                {
                    TileType ta = grid[x, y].type;
                    for (int d = 0; d < 2; d++)
                    {
                        int nx = x + ddx[d], ny = y + ddy[d];
                        if (nx >= width || ny >= height) continue;
                        TileType tb = grid[nx, ny].type;
                        if (ta != tb)
                        {
                            var pair = ta < tb ? (ta, tb) : (tb, ta);
                            adjPairs.Add(pair);
                        }
                    }
                }
            }

            // 6. Generate transition texture per adjacency pair — parallel
            var transitionTasks = new Dictionary<(TileType, TileType), Task<Texture2D>>();
            foreach (var pair in adjPairs)
            {
                string cacheKey = $"{pair.Item1}-{pair.Item2}:{primaryKeyword}";
                Texture2D texA = tileTextures.ContainsKey(pair.Item1) ? tileTextures[pair.Item1] : styleAnchor;
                Texture2D texB = tileTextures.ContainsKey(pair.Item2) ? tileTextures[pair.Item2] : styleAnchor;
                string prompt = $"Seamless top-down 2D RPG transition tile where {primaryKeyword} " +
                                $"{pair.Item1.ToString().ToLower()} blends into {pair.Item2.ToString().ToLower()}";
                transitionTasks[pair] = LLMService.Instance.GenerateTransitionTile(prompt, texA, texB, cacheKey);
            }
            await Task.WhenAll(transitionTasks.Values);
            foreach (var kvp in transitionTasks)
                transitionTextures[kvp.Key] = kvp.Value.Result ?? Texture2D.whiteTexture;

            // 7. Instantiate tile GameObjects
            for (int x = 0; x < width; x++)
                for (int y = 0; y < height; y++)
                    CreateTileVisual(x, y, GetTileTexture(x, y));

            // 8. ASCII map log
            string mapLog = "Generated Map:\n";
            for (int y = height - 1; y >= 0; y--)
            {
                for (int x = 0; x < width; x++)
                {
                    switch (grid[x, y].type)
                    {
                        case TileType.Wall:       mapLog += "[W]"; break;
                        case TileType.Water:      mapLog += "[~]"; break;
                        case TileType.Path:       mapLog += "[.]"; break;
                        case TileType.Door:       mapLog += "[D]"; break;
                        case TileType.Chest:      mapLog += "[C]"; break;
                        case TileType.EnemySpawn: mapLog += "[E]"; break;
                        case TileType.Exit:       mapLog += "[X]"; break;
                        default:                  mapLog += "[ ]"; break;
                    }
                }
                mapLog += "\n";
            }
            Debug.Log(mapLog);

            Debug.Log("Map Generation Completed");
            if (DNDLLM.Core.GameManager.Instance != null)
                DNDLLM.Core.GameManager.Instance.ChangeState(DNDLLM.Core.GameState.CharacterGeneration);

            AdjustCamera();
        }
```
